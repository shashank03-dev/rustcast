3/// Fuzzy-rank every app against the query.
4fn rank(apps: &[App], query: &str) -> Vec<(u32, &App)> {
5    let started = Instant::now();
6    let mut hits: Vec<_> = apps
7        .iter()
8        .filter_map(|app| score(&app.name, query).map(|s| (s, app)))
9        .collect();
10
11    hits.sort_by(|a, b| b.0.cmp(&a.0));
12    println!("ranked {} apps in {:?}", hits.len(), started.elapsed());
13    hits
